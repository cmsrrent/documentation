Git Beginner Cheat Sheet

1. Set up Git
	git --help
	git config --list
	git config --global user.name "Your Name"
	git config --global user.email "you@example.com"
	git remote -v

2. Start a new project
	cd your-project-folder
	git init
	git remote add origin https://github.com/your-user/your-repo.git

3. Clone an existing repository
	git clone https://github.com/your-user/your-repo.git
	cd your-repo

4. Check status
	git status
	git status -sb

5. Create and switch branches
	git branch
	git checkout -b feature/login
	git switch -c feature/login
	git checkout main
	git switch main

6. Make changes
	git add file-name.txt
	git add .
	git add -A
	git commit -m "Add login form"

7. Push to GitHub
	git push -u origin main
	git push origin main
	git push

8. Pull updates from GitHub
	git pull
	git pull --rebase origin main
	git fetch origin

9. View history
	git log
	git log --oneline
	git log --oneline --decorate --graph --all
	git show <commit-hash>

10. Compare changes
	git diff
	git diff --staged

11. Undo / fix mistakes
	git restore <file>
	git restore --staged <file>
	git reset HEAD~1
	git reset --hard HEAD
	git revert <commit-hash>

12. Save work temporarily
	git stash
	git stash list
	git stash pop
	git stash apply stash@{0}

13. Ignore files
	touch .gitignore
	Add entries like:
		log.txt
		build/
		*.log
		.env
	git rm --cached <file>

14. Delete and merge branches
	git branch -d feature/login
	git branch -D feature/login
	git merge main
	git merge feature/login

15. Useful remote commands
	git remote add origin <url>
	git remote remove origin
	git branch -r
	git checkout --track origin/feature/login
	git push origin --delete feature/login

Quick workflow
	1. git status
	2. git add .
	3. git commit -m "Your message"
	4. git push

Notes
	- Use main or master depending on the project.
	- Always check git status before pulling or pushing.
	- git reset --hard and git clean -fd can delete work permanently, so use carefully.
	- git pull combines fetching and merging; git fetch only downloads changes.
